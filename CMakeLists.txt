cmake_minimum_required(VERSION 3.16)
project(abw VERSION 0.1.0 LANGUAGES CXX)

find_path(ABW_BOOST_INCLUDE_DIR boost/sort/pdqsort/pdqsort.hpp
    HINTS ${BOOST_ROOT} ENV BOOST_ROOT PATH_SUFFIXES include)
if(NOT ABW_BOOST_INCLUDE_DIR)
    message(FATAL_ERROR "Boost headers required. Set ABW_BOOST_INCLUDE_DIR to your Boost root.")
endif()
add_library(abw INTERFACE)
add_library(abw::abw ALIAS abw)
target_compile_features(abw INTERFACE cxx_std_17)
target_include_directories(abw INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(abw SYSTEM INTERFACE ${ABW_BOOST_INCLUDE_DIR})

add_executable(abw_example examples/basic.cpp)
target_link_libraries(abw_example PRIVATE abw::abw)
option(ABW_BUILD_TESTS "Build frozen-policy conformance tests" ON)
if(ABW_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(abw_conformance tests/conformance.cpp)
    target_link_libraries(abw_conformance PRIVATE abw::abw Threads::Threads)
    target_include_directories(abw_conformance PRIVATE tests/reference)
    add_test(NAME abw_conformance COMMAND abw_conformance)
endif()
